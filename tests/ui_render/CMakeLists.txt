cmake_minimum_required(VERSION 3.16)
project(badge_ui_render C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/managed_components/lvgl__lvgl" lvgl)
option(BADGE_UI_CLIB_ALLOCATOR "Use firmware's shared C allocator" OFF)
if(BADGE_UI_CLIB_ALLOCATOR)
    target_compile_definitions(lvgl PUBLIC BADGE_UI_USE_CLIB_ALLOCATOR=1)
endif()
file(GLOB BADGE_FONTS "${REPO}/assets/fonts/github-badge/badge_font_*.c")
add_executable(badge_ui_render render.c "${REPO}/main/badge_ui.c" "${REPO}/main/badge_ui_state.c" "${REPO}/main/badge_model.c" "${REPO}/main/badge_sync_parse.c" "${REPO}/components/bsp/src/bsp_display_rounding.c" ${BADGE_FONTS})
target_include_directories(badge_ui_render PRIVATE "${REPO}/main" "${REPO}/components/bsp/src")
target_link_libraries(badge_ui_render PRIVATE lvgl)
if(UNIX)
    target_link_libraries(badge_ui_render PRIVATE m)
endif()
target_compile_options(badge_ui_render PRIVATE -Wall -Wextra -Werror)
