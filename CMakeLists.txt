cmake_minimum_required(VERSION 3.16)
if(DEFINED ENV{BADGE_VERSION} AND NOT "$ENV{BADGE_VERSION}" STREQUAL "")
    set(BADGE_VERSION "$ENV{BADGE_VERSION}")
else()
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" BADGE_VERSION LIMIT_COUNT 1)
endif()
if(NOT BADGE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "BADGE_VERSION must be X.Y.Z without leading zeros or suffixes")
endif()
set(PROJECT_VER "github-badge-${BADGE_VERSION}")
string(LENGTH "${PROJECT_VER}" BADGE_VERSION_LENGTH)
if(BADGE_VERSION_LENGTH GREATER 31)
    message(FATAL_ERROR "BADGE_VERSION exceeds application descriptor capacity")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(FoloToy-AI-Passport)
